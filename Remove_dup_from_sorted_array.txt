package LeetCode_Again;

public class remove_duplicate_from_sorted_array_26 {
	
	public static int removeele(int num[])
	{
		int count=0;
		for(int i=0;i<num.length;i++)
		{
			if(i<num.length-1 && num[i]==num[i+1])
			{
				continue;
			}
			else
			{
				num[count]=num[i];
				count++;
			}
		}
		
		return count;
	}
	
	public static void main(String args[])
	{
		int arr[]= {1,1,2,2,4,5,5,6};
		System.out.println(removeele(arr));
	}

}
